cmake_minimum_required(VERSION 3.24)
project(orca_l0 VERSION 0.1.0 LANGUAGES CXX)

option(ORCA_L0_BUILD_RUNTIME "Build the Level Zero runtime" ON)
option(ORCA_L0_BUILD_CHECKS "Build the standalone checks" ${PROJECT_IS_TOP_LEVEL})
option(ORCA_L0_BUILD_GPU_CHECKS "Compile kernels and register GPU checks" ${ORCA_L0_BUILD_CHECKS})
option(ORCA_L0_DIRECT_DRIVER "Bind the Intel driver DDI directly, without ze_loader" OFF)
option(ORCA_L0_BUILD_DRM_CHECKS "Build the experimental Linux DG2 i915 runtime and checks" OFF)
option(ORCA_L0_BUILD_WDDM_CHECKS "Build the Windows direct D3DKMT check (no vendor UMD)" OFF)
option(ORCA_L0_BUILD_WDDM_BACKEND "Build the experimental Windows DG2 runtime (no vendor UMD)" OFF)
set(ORCA_L0_DEVICE "dg2" CACHE STRING "ocloc target for the check kernels")
set(ORCA_L0_SDK "" CACHE PATH "Level Zero SDK root")
set(ORCA_L0_WDDM_ZEBIN "" CACHE FILEPATH "Optional inference.cl zebin for the Windows direct compute checks")
set(ORCA_L0_DG2_MEMORY_ZEBIN "" CACHE FILEPATH "Optional driver-baked src/drm/memory.cl zebin for native transfer checks")

if(ORCA_L0_BUILD_CHECKS)
  enable_testing()
  find_package(Threads REQUIRED)
  add_executable(orca-l0-ring-check tests/ring_check.cpp)
  target_include_directories(orca-l0-ring-check PRIVATE include)
  target_compile_features(orca-l0-ring-check PRIVATE cxx_std_17)
  target_link_libraries(orca-l0-ring-check PRIVATE Threads::Threads)
  add_test(NAME orca-l0-ring COMMAND orca-l0-ring-check)
  add_executable(orca-l0-blitter-check tests/blitter_check.cpp)
  target_compile_features(orca-l0-blitter-check PRIVATE cxx_std_17)
  add_test(NAME orca-l0-blitter-packets COMMAND orca-l0-blitter-check)
  set_tests_properties(orca-l0-blitter-packets PROPERTIES TIMEOUT 30)
  set_tests_properties(orca-l0-ring PROPERTIES TIMEOUT 30)
endif()

if(ORCA_L0_BUILD_DRM_CHECKS)
  find_package(Threads REQUIRED)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The DRM submission check requires Linux")
  endif()
  find_path(ORCA_L0_DRM_INCLUDE_DIR i915_drm.h PATH_SUFFIXES libdrm REQUIRED)
  add_executable(orca-l0-drm-submit-check tests/drm_submit_check.cpp)
  target_compile_features(orca-l0-drm-submit-check PRIVATE cxx_std_17)
  target_compile_options(orca-l0-drm-submit-check PRIVATE -Wall -Wextra -Wpedantic)
  target_include_directories(orca-l0-drm-submit-check SYSTEM PRIVATE "${ORCA_L0_DRM_INCLUDE_DIR}")
  add_library(orca_dg2 STATIC src/drm/runtime.cpp)
  add_library(Orca::DG2 ALIAS orca_dg2)
  target_link_libraries(orca_dg2 PRIVATE Threads::Threads)
  target_compile_features(orca_dg2 PUBLIC cxx_std_17)
  target_include_directories(orca_dg2 PUBLIC include)
  target_include_directories(orca_dg2 SYSTEM PRIVATE "${ORCA_L0_DRM_INCLUDE_DIR}" third_party)
  target_compile_options(orca_dg2 PRIVATE -Wall -Wextra)
  add_executable(orca-l0-drm-compute-check tests/drm_compute_check.cpp)
  target_link_libraries(orca-l0-drm-compute-check PRIVATE Orca::DG2)
  if(ORCA_L0_BUILD_CHECKS)
    add_test(NAME orca-l0-drm-packets COMMAND orca-l0-drm-submit-check --self-check)
    add_test(NAME orca-l0-drm-compute-cpu COMMAND orca-l0-drm-compute-check --self-check)
    if(ORCA_L0_DG2_MEMORY_ZEBIN)
      add_test(NAME orca-l0-drm-transfers COMMAND orca-l0-drm-compute-check --gpu-transfers "${ORCA_L0_DG2_MEMORY_ZEBIN}")
      set_tests_properties(orca-l0-drm-transfers PROPERTIES LABELS gpu TIMEOUT 90 RUN_SERIAL TRUE)
    endif()
  endif()
endif()

if(ORCA_L0_BUILD_WDDM_BACKEND)
  find_package(Threads REQUIRED)
  if(NOT WIN32)
    message(FATAL_ERROR "The WDDM backend requires Windows")
  endif()
  add_library(orca_dg2 STATIC src/wddm/runtime.cpp)
  add_library(Orca::DG2 ALIAS orca_dg2)
  target_compile_features(orca_dg2 PUBLIC cxx_std_17)
  target_include_directories(orca_dg2 PUBLIC include)
  target_include_directories(orca_dg2 PRIVATE third_party)
  target_link_libraries(orca_dg2 PRIVATE gdi32 Threads::Threads)
  add_executable(orca-l0-wddm-allocation-check tests/wddm_allocation_check.cpp)
  target_compile_features(orca-l0-wddm-allocation-check PRIVATE cxx_std_17)
  target_include_directories(orca-l0-wddm-allocation-check PRIVATE include third_party)
  target_link_libraries(orca-l0-wddm-allocation-check PRIVATE gdi32 Threads::Threads)
  if(ORCA_L0_BUILD_CHECKS)
    add_test(NAME orca-l0-dg2-allocation-recovery COMMAND orca-l0-wddm-allocation-check)
    set_tests_properties(orca-l0-dg2-allocation-recovery PROPERTIES LABELS gpu TIMEOUT 90 RUN_SERIAL TRUE)
  endif()
  add_executable(orca-l0-wddm-import-check tests/wddm_import_check.cpp)
  target_compile_features(orca-l0-wddm-import-check PRIVATE cxx_std_17)
  target_include_directories(orca-l0-wddm-import-check PRIVATE include third_party)
  target_link_libraries(orca-l0-wddm-import-check PRIVATE gdi32 Threads::Threads)
  if(ORCA_L0_BUILD_CHECKS AND ORCA_L0_DG2_MEMORY_ZEBIN)
    add_test(NAME orca-l0-dg2-host-import COMMAND orca-l0-wddm-import-check "${ORCA_L0_DG2_MEMORY_ZEBIN}")
    set_tests_properties(orca-l0-dg2-host-import PROPERTIES LABELS gpu TIMEOUT 180 RUN_SERIAL TRUE)
  endif()
  add_executable(orca-l0-wddm-host-check tests/wddm_host_check.cpp)
  target_compile_features(orca-l0-wddm-host-check PRIVATE cxx_std_17)
  target_include_directories(orca-l0-wddm-host-check PRIVATE include third_party)
  target_link_libraries(orca-l0-wddm-host-check PRIVATE gdi32 Threads::Threads)
  if(ORCA_L0_BUILD_CHECKS)
    add_test(NAME orca-l0-dg2-host-ownership COMMAND orca-l0-wddm-host-check)
    set_tests_properties(orca-l0-dg2-host-ownership PROPERTIES LABELS gpu TIMEOUT 90 RUN_SERIAL TRUE)
  endif()
  add_executable(orca-l0-dg2-check tests/drm_compute_check.cpp)
  target_link_libraries(orca-l0-dg2-check PRIVATE Orca::DG2)
  add_executable(orca-l0-dg2-bench EXCLUDE_FROM_ALL tests/wddm_memory_check.cpp)
  target_compile_definitions(orca-l0-dg2-bench PRIVATE ORCA_L0_WDDM_DG2=1)
  target_include_directories(orca-l0-dg2-bench PRIVATE third_party)
  target_link_libraries(orca-l0-dg2-bench PRIVATE Orca::DG2 gdi32)
  if(ORCA_L0_BUILD_CHECKS)
    add_test(NAME orca-l0-dg2-cpu COMMAND orca-l0-dg2-check --self-check)
    if(ORCA_L0_DG2_MEMORY_ZEBIN)
      add_test(NAME orca-l0-dg2-transfers COMMAND orca-l0-dg2-check --gpu-transfers "${ORCA_L0_DG2_MEMORY_ZEBIN}")
      set_tests_properties(orca-l0-dg2-transfers PROPERTIES LABELS gpu TIMEOUT 90 RUN_SERIAL TRUE)
      add_test(NAME orca-l0-dg2-large-memory COMMAND orca-l0-dg2-check --gpu-large-device-memory "${ORCA_L0_DG2_MEMORY_ZEBIN}")
      set_tests_properties(orca-l0-dg2-large-memory PROPERTIES LABELS gpu TIMEOUT 90 RUN_SERIAL TRUE)
    endif()
    if(ORCA_L0_WDDM_ZEBIN)
      if(ORCA_L0_DG2_MEMORY_ZEBIN)
        add_test(NAME orca-l0-dg2-multilist COMMAND orca-l0-dg2-check --gpu-multilist "${ORCA_L0_WDDM_ZEBIN}" "${ORCA_L0_DG2_MEMORY_ZEBIN}")
        add_test(NAME orca-l0-dg2-reclaim COMMAND orca-l0-dg2-check --gpu-reclaim "${ORCA_L0_WDDM_ZEBIN}" "${ORCA_L0_DG2_MEMORY_ZEBIN}")
        add_test(NAME orca-l0-dg2-multiqueue COMMAND orca-l0-dg2-check --gpu-multiqueue "${ORCA_L0_WDDM_ZEBIN}" "${ORCA_L0_DG2_MEMORY_ZEBIN}")
        add_test(NAME orca-l0-dg2-host-gate COMMAND orca-l0-dg2-check --gpu-host-gate "${ORCA_L0_WDDM_ZEBIN}" "${ORCA_L0_DG2_MEMORY_ZEBIN}")
        add_test(NAME orca-l0-dg2-inline-events COMMAND orca-l0-dg2-check --gpu-inline-events "${ORCA_L0_WDDM_ZEBIN}" "${ORCA_L0_DG2_MEMORY_ZEBIN}")
        add_test(NAME orca-l0-dg2-observers COMMAND orca-l0-dg2-check --gpu-observers "${ORCA_L0_WDDM_ZEBIN}" "${ORCA_L0_DG2_MEMORY_ZEBIN}")
        add_test(NAME orca-l0-dg2-recording COMMAND orca-l0-dg2-check --gpu-recording "${ORCA_L0_WDDM_ZEBIN}" "${ORCA_L0_DG2_MEMORY_ZEBIN}")
        add_test(NAME orca-l0-dg2-release COMMAND orca-l0-dg2-check --gpu-release "${ORCA_L0_WDDM_ZEBIN}" "${ORCA_L0_DG2_MEMORY_ZEBIN}")
        set_tests_properties(orca-l0-dg2-multilist PROPERTIES LABELS gpu TIMEOUT 90 RUN_SERIAL TRUE)
        set_tests_properties(orca-l0-dg2-reclaim PROPERTIES LABELS gpu TIMEOUT 90 RUN_SERIAL TRUE)
        set_tests_properties(orca-l0-dg2-multiqueue PROPERTIES LABELS gpu TIMEOUT 90 RUN_SERIAL TRUE)
        set_tests_properties(orca-l0-dg2-host-gate PROPERTIES LABELS gpu TIMEOUT 90 RUN_SERIAL TRUE)
        set_tests_properties(orca-l0-dg2-inline-events PROPERTIES LABELS gpu TIMEOUT 90 RUN_SERIAL TRUE)
        set_tests_properties(orca-l0-dg2-observers PROPERTIES LABELS gpu TIMEOUT 90 RUN_SERIAL TRUE)
        set_tests_properties(orca-l0-dg2-recording PROPERTIES LABELS gpu TIMEOUT 90 RUN_SERIAL TRUE)
        set_tests_properties(orca-l0-dg2-release PROPERTIES LABELS gpu TIMEOUT 90 RUN_SERIAL TRUE)
      endif()
      add_test(NAME orca-l0-dg2-gpu COMMAND orca-l0-dg2-check --gpu "${ORCA_L0_WDDM_ZEBIN}")
      add_test(NAME orca-l0-dg2-semisync COMMAND orca-l0-dg2-check --gpu-semisync "${ORCA_L0_WDDM_ZEBIN}")
      add_test(NAME orca-l0-dg2-memory COMMAND orca-l0-dg2-check --gpu-memory "${ORCA_L0_WDDM_ZEBIN}")
      set_tests_properties(orca-l0-dg2-gpu orca-l0-dg2-semisync orca-l0-dg2-memory PROPERTIES LABELS gpu TIMEOUT 90 RUN_SERIAL TRUE)
    endif()
  endif()
endif()

if(ORCA_L0_BUILD_WDDM_CHECKS)
  if(NOT WIN32)
    message(FATAL_ERROR "The WDDM memory check requires Windows")
  endif()
  add_executable(orca-l0-wddm-memory-check tests/wddm_memory_check.cpp)
  target_compile_features(orca-l0-wddm-memory-check PRIVATE cxx_std_17)
  target_include_directories(orca-l0-wddm-memory-check PRIVATE third_party)
  target_link_libraries(orca-l0-wddm-memory-check PRIVATE gdi32)
  if(ORCA_L0_BUILD_CHECKS)
    add_test(NAME orca-l0-wddm-memory COMMAND orca-l0-wddm-memory-check)
    add_test(NAME orca-l0-wddm-map COMMAND orca-l0-wddm-memory-check --map)
    add_test(NAME orca-l0-wddm-context COMMAND orca-l0-wddm-memory-check --context)

    if(ORCA_L0_WDDM_ZEBIN)
      if(NOT EXISTS "${ORCA_L0_WDDM_ZEBIN}")
        message(FATAL_ERROR "ORCA_L0_WDDM_ZEBIN does not exist")
      endif()
      add_test(NAME orca-l0-wddm-compute COMMAND orca-l0-wddm-memory-check --compute "${ORCA_L0_WDDM_ZEBIN}")

    endif()
  endif()

if(NOT ORCA_L0_BUILD_RUNTIME)
  return()
endif()
find_path(ORCA_L0_INCLUDE_DIR level_zero/ze_api.h
  HINTS "${ORCA_L0_SDK}/include" "$ENV{LEVEL_ZERO_ROOT}/include")
if(NOT ORCA_L0_DIRECT_DRIVER)
  find_library(ORCA_L0_LOADER NAMES ze_loader
    HINTS "${ORCA_L0_SDK}/lib" "${ORCA_L0_SDK}/lib/x64"
          "$ENV{LEVEL_ZERO_ROOT}/lib" "$ENV{LEVEL_ZERO_ROOT}/lib/x64")
endif()
if(NOT ORCA_L0_INCLUDE_DIR OR (NOT ORCA_L0_DIRECT_DRIVER AND NOT ORCA_L0_LOADER))
  message(FATAL_ERROR "Level Zero headers/loader missing; set ORCA_L0_SDK (CPU ring check: ORCA_L0_BUILD_RUNTIME=OFF)")
endif()
add_library(orca_l0 STATIC src/runtime.cpp)
add_library(Orca::L0 ALIAS orca_l0)
target_compile_features(orca_l0 PUBLIC cxx_std_17)
target_include_directories(orca_l0 PUBLIC include "${ORCA_L0_INCLUDE_DIR}")
if(ORCA_L0_DIRECT_DRIVER)
  find_package(Threads REQUIRED)
  target_sources(orca_l0 PRIVATE src/direct_driver.cpp)
  target_compile_definitions(orca_l0 PUBLIC ORCA_L0_DIRECT_DRIVER=1)
  target_link_libraries(orca_l0 PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
else()
  target_link_libraries(orca_l0 PUBLIC "${ORCA_L0_LOADER}")
endif()
add_executable(orca-l0-native-pack EXCLUDE_FROM_ALL compiler/native_pack.cpp)
target_link_libraries(orca-l0-native-pack PRIVATE Orca::L0)
if(WIN32 AND ORCA_L0_BUILD_WDDM_CHECKS AND ORCA_L0_DIRECT_DRIVER)
  add_executable(orca-l0-wddm-baseline-check EXCLUDE_FROM_ALL tests/wddm_memory_check.cpp)
  target_compile_definitions(orca-l0-wddm-baseline-check PRIVATE ORCA_L0_WDDM_BASELINE=1)
  target_include_directories(orca-l0-wddm-baseline-check PRIVATE third_party)
  target_link_libraries(orca-l0-wddm-baseline-check PRIVATE Orca::L0 gdi32)
endif()
if(MSVC)
  target_compile_options(orca_l0 PRIVATE /W4)
else()
  target_compile_options(orca_l0 PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(ORCA_L0_BUILD_CHECKS)
  add_executable(orca-l0-check tests/runtime_check.cpp)
  target_link_libraries(orca-l0-check PRIVATE Orca::L0 Threads::Threads)
  add_executable(orca-l0-submit-check EXCLUDE_FROM_ALL tests/submit_check.cpp)
  target_link_libraries(orca-l0-submit-check PRIVATE Orca::L0)
  add_executable(orca-l0-native-pack-check tests/native_pack_check.cpp)
  target_link_libraries(orca-l0-native-pack-check PRIVATE Orca::L0)
  add_dependencies(orca-l0-native-pack-check orca-l0-native-pack)
  add_test(NAME orca-l0-native-pack-preflight COMMAND orca-l0-native-pack --self-check)
endif()
if(ORCA_L0_BUILD_CHECKS AND ORCA_L0_BUILD_GPU_CHECKS)
  find_program(ORCA_L0_OCLOC NAMES ocloc)
  if(NOT ORCA_L0_OCLOC)
    message(FATAL_ERROR "ocloc missing; set ORCA_L0_OCLOC or disable ORCA_L0_BUILD_GPU_CHECKS")
  endif()
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/inference.spv" "${CMAKE_CURRENT_BINARY_DIR}/inference.bin"
    COMMAND "${ORCA_L0_OCLOC}" -file "${CMAKE_CURRENT_SOURCE_DIR}/tests/inference.cl"
            -device "${ORCA_L0_DEVICE}" -output inference -output_no_suffix
            -out_dir "${CMAKE_CURRENT_BINARY_DIR}"
    DEPENDS tests/inference.cl VERBATIM)
  add_custom_target(orca-l0-check-kernels DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/inference.spv"
                                               "${CMAKE_CURRENT_BINARY_DIR}/inference.bin")
  add_dependencies(orca-l0-check orca-l0-check-kernels)
  add_test(NAME orca-l0-runtime COMMAND orca-l0-check "${CMAKE_CURRENT_BINARY_DIR}/inference.spv")
  add_test(NAME orca-l0-poison COMMAND orca-l0-check "${CMAKE_CURRENT_BINARY_DIR}/inference.spv")
  set_tests_properties(orca-l0-poison PROPERTIES ENVIRONMENT "ORCA_L0_POISON_ALLOCATIONS=1"
    LABELS gpu TIMEOUT 120 RUN_SERIAL TRUE)
  add_test(NAME orca-l0-native COMMAND orca-l0-check "${CMAKE_CURRENT_BINARY_DIR}/inference.bin" native)
  set_tests_properties(orca-l0-runtime orca-l0-native PROPERTIES LABELS gpu TIMEOUT 120 RUN_SERIAL TRUE)
endif()
